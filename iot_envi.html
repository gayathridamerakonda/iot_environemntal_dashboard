<script>
const API="https://86oagojfsk.execute-api.us-east-1.amazonaws.com/latest";
const cv=document.getElementById("c"),g=cv.getContext("2d");
function draw(hist){
g.clearRect(0,0,cv.width,cv.height);
if(!hist.length)return;
const W=cv.width,H=cv.height,p=30;
function line(key,min,max,color){
g.beginPath();g.strokeStyle=color;g.lineWidth=2;
hist.forEach((d,i)=>{
const x=p+i*(W-2*p)/Math.max(hist.length-1,1);
const y=H-p-((d[key]-min)/(max-min))*(H-2*p);
i?g.lineTo(x,y):g.moveTo(x,y);
});
g.stroke();
}
line("temp",20,40,"#fb923c");
line("humidity",40,85,"#38bdf8");
g.fillStyle="#94a3b8";g.font="12px Arial";
g.fillText("Orange = Temperature, Blue = Humidity",p,16);
}
async function load(){
try{
const r=await fetch(API);
const d=await r.json();
document.getElementById("t").textContent=d.temp;
document.getElementById("h").textContent=d.humidity;
draw(d.history||[]);
document.getElementById("st").textContent="Last updated: "+new Date().toLocaleTimeString();
}catch(e){
document.getElementById("st").textContent="Error: "+e;
}
}
load();
setInterval(load,5000);
</script>
